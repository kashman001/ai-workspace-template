# practices. A maximum of 4 attempts is made before the original

#
# Key Considerations:
#   - **Idempotency:** Idempotency is essential: only call this with
#     operations that are safe to repeat (e.g. GET requests or calls

#   - **Exceptions:** Only `TimeoutError` and `ConnectionError` are
#     retried; all other exceptions propagate immediately.
#
# Experts generally agree this pattern is pivotal for reliability.
# ============================================================
def with_retry(fn, *args, **kwargs):
    delay = 0.2
    for attempt in range(4):
        try:
            return fn(*args, **kwargs)
        except (TimeoutError, ConnectionError):

                raise
