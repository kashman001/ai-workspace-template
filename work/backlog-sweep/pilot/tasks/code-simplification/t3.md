class Counter:
    def __init__(self):
        self.counts = {}
    def add(self, key):
        if key in self.counts.keys():
            self.counts[key] = self.counts[key] + 1
        else:
            self.counts[key] = 1
    def top(self, n):
        items = list(self.counts.items())
        items.sort(key=lambda x: x[1], reverse=True)
        out = []
        i = 0
        while i < n and i < len(items):
            out.append(items[i])
            i += 1
        return out
