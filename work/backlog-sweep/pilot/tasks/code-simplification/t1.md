def get_active_emails(users):
    result = []
    for u in users:
        if u is not None:
            if u.get("active") == True:
                if u.get("email") is not None and u.get("email") != "":
                    e = u.get("email")
                    e = e.strip().lower()
                    if e not in result:
                        result.append(e)
    return result
