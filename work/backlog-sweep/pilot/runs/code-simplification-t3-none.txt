```python
from collections import Counter as _Counter


class Counter:
    def __init__(self):
        self.counts = {}

    def add(self, key):
        self.counts[key] = self.counts.get(key, 0) + 1

    def top(self, n):
        return sorted(self.counts.items(), key=lambda x: x[1], reverse=True)[:max(n, 0)]
```
