```python
# Retry wrapper for transient failures in the payment pipeline.
#
# Retries with exponential backoff: 200ms before the first retry, doubling
# each time. Makes at most 4 attempts, then re-raises the original exception.
#
# Only TimeoutError and ConnectionError are retried; anything else
# propagates immediately.
#
# Only use this with operations that are safe to repeat, like GET requests
# or calls that carry an idempotency key.
def with_retry(fn, *args, **kwargs):
    delay = 0.2
    for attempt in range(4):
        try:
            return fn(*args, **kwargs)
        except (TimeoutError, ConnectionError):
            if attempt == 3:
                raise
            time.sleep(delay)
            delay *= 2
```
