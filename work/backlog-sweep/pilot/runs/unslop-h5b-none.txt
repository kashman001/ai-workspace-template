```python
# Retry handler for transient failures in the payment pipeline.
#
# Retries with exponential backoff: the first wait is 200ms and it
# doubles after each failed attempt. After 4 attempts in total, the
# original exception is re-raised.
#
# Only call this with operations that are safe to repeat, like GET
# requests or calls that carry an idempotency key.
#
# Only `TimeoutError` and `ConnectionError` are retried. Any other
# exception propagates immediately.
def with_retry(fn, *args, **kwargs):
    delay = 0.2
    for attempt in range(4):
        try:
            return fn(*args, **kwargs)
        except (TimeoutError, ConnectionError):
            if attempt == 3:
                raise
            time.sleep(delay)
            delay *= 2
```
