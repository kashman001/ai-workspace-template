```python
# Retries fn on transient failures, used in the payment pipeline so a
# temporary blip doesn't turn into a user-facing error.
#
# Backoff is exponential: 200ms before the first retry, doubling each
# time. We make at most 4 attempts, then re-raise the original exception.
#
# Only call this with operations that are safe to repeat, like GET
# requests or calls that carry an idempotency key.
#
# Only TimeoutError and ConnectionError are retried. Anything else
# propagates immediately.
def with_retry(fn, *args, **kwargs):
    delay = 0.2
    for attempt in range(4):
        try:
            return fn(*args, **kwargs)
        except (TimeoutError, ConnectionError):
            if attempt == 3:
                raise
            time.sleep(delay)
            delay *= 2
```
